from pathlib import Path
import subprocess,json
root=Path.home()/'club-system-test/backend'
base=json.loads(subprocess.check_output(['podman','inspect','club-capacity-test_nginx_1']))[0]
assert base['Config']['Labels']['com.docker.compose.project']=='club-capacity-test'
assert not subprocess.check_output(['ss','-H','-lnt','sport = :18083'],text=True).strip()
source=(root/'.local/nginx.test.tls.conf').read_text().split('map $http_upgrade')[0]
source+='''map $http_upgrade $connection_upgrade { default upgrade; '' ''; }
upstream probe_http {
 server 192.168.100.130:18081;
 keepalive 128;
 keepalive_timeout 60s;
}
server {
 listen 192.168.100.130:18083 ssl backlog=2048;
 ssl_certificate /etc/nginx/test-tls/test.crt;
 ssl_certificate_key /etc/nginx/test-tls/test.key;
 ssl_protocols TLSv1.2 TLSv1.3;
 ssl_session_cache shared:club_tls:10m;
 access_log /tmp/club-edge-requests.jsonl club_diag buffer=64k flush=1s;
 location / {
  proxy_pass http://probe_http;
  proxy_http_version 1.1;
  proxy_set_header Connection $connection_upgrade;
  proxy_set_header Upgrade $http_upgrade;
  proxy_read_timeout 70s;
 }
}
'''
p=root/'.local/edge-probe.conf';p.write_text(source)
subprocess.run(['podman','run','-d','--name','club-capacity-test_edge_probe','--network','host','--cgroup-parent',base['HostConfig']['CgroupParent'],'--memory','128m','--label','club.diagnostic=true','-v',str(root/'deploy/nginx.conf')+':/etc/nginx/nginx.conf:ro','-v',str(p)+':/etc/nginx/conf.d/default.conf:ro','-v',str(root/'.local/tls-ecdsa')+':/etc/nginx/test-tls:ro',base['ImageName']],check=True)
