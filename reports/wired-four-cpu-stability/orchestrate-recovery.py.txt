import asyncio
import json
from pathlib import Path
import shlex
import time

ROOT = Path('reports/wired-four-cpu-stability')
TARGET = ['ssh', '-S', '/private/tmp/club-test-current-ssh', '-o', 'BatchMode=yes', 'gg1214@192.168.100.130']
LOAD = ['ssh', '-S', '/private/tmp/club-load20-ssh', '-o', 'BatchMode=yes', 'GG1214@192.168.100.20']

async def run(argv):
    p = await asyncio.create_subprocess_exec(*argv, stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE)
    out, err = await p.communicate()
    if p.returncode:
        raise RuntimeError(f'Command failed {p.returncode}: {err.decode()[-300:]}')
    return out.decode()

async def target(cmd):
    return await run(TARGET + ['cd ~/club-system-test/backend && ' + cmd])

async def monitor(stop):
    with (ROOT/'recovery-process-health.jsonl').open('w') as out:
        while not stop.is_set():
            try:
                sample = json.loads(await target('python3 deploy/stability-test-snapshot.py'))
            except Exception as exc:
                sample = {'at': time.time(), 'error': type(exc).__name__}
            out.write(json.dumps(sample)+'\n'); out.flush()
            try:
                await asyncio.wait_for(stop.wait(), 5)
            except TimeoutError:
                pass

async def case(name, mode, scenario, rounds, hold):
    args = ['.venv/bin/python', 'scripts/load_distributed_journey.py', '--url', 'https://192.168.100.130:18082', '--ca-cert', 'private/test-ca.crt', '--ssh', 'gg1214@192.168.100.130', '--control', '/tmp/club-target-test-ssh', '--users', '1500', '--processes', '3', '--client-profile', 'connected_first', '--mode', mode, '--scenario', scenario, '--rounds', str(rounds), '--hold-seconds', str(hold), '--report', f'reports/{name}.json']
    print('START '+name, flush=True)
    p = await asyncio.create_subprocess_exec(*LOAD, 'cd ~/club-load-test && ulimit -n 16384 && '+shlex.join(args))
    code = await p.wait()
    if code == 0:
        recovery_args=['.venv/bin/python','scripts/recover_load_students.py','--journey-report',f'reports/{name}.json','--report',f'reports/{name}-recovery.json','--ca-cert','private/test-ca.crt']
        print(await run(LOAD+['cd ~/club-load-test && '+shlex.join(recovery_args)]),flush=True)
        await run(['scp','-C','-o','ControlPath=/private/tmp/club-load20-ssh','-o','BatchMode=yes',f'GG1214@192.168.100.20:club-load-test/reports/{name}-recovery.json',str(ROOT)])
    await run(['scp', '-C', '-o', 'ControlPath=/private/tmp/club-load20-ssh', '-o', 'BatchMode=yes', f'GG1214@192.168.100.20:club-load-test/reports/{name}.json', str(ROOT)])
    parts = ROOT/f'{name}.parts'; parts.mkdir()
    await run(['scp', '-C', '-o', 'ControlPath=/private/tmp/club-load20-ssh', '-o', 'BatchMode=yes', f'GG1214@192.168.100.20:club-load-test/reports/{name}.parts/*.json', str(parts)])
    if code:
        raise RuntimeError(f'Journey failed {code}; report retained')
    print('DONE '+name, flush=True)

async def main():
    assert (ROOT/'after.json').exists(), 'Original stability run must finish first'
    (ROOT/'recovery-before.json').write_text(await target('python3 deploy/metrics-test.py'))
    stop=asyncio.Event(); task=asyncio.create_task(monitor(stop))
    try:
        for n in range(1,4):
            await case(f'four-cpu-retry-{n}', 'login', 'uniform', 1, 45)
    finally:
        stop.set(); await task
        (ROOT/'recovery-after.json').write_text(await target('python3 deploy/metrics-test.py'))
        (ROOT/'recovery-final-processes.json').write_text(await target('python3 deploy/stability-test-snapshot.py'))

asyncio.run(main())
